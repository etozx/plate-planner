function cascadeIn(els) {
  if (reduced()) return;
  els.forEach((el, i) => { el.classList.remove('cascade'); el.style.setProperty('--i', Math.min(i, 14)); void el.offsetWidth; el.classList.add('cascade'); });
}

function closeDropdowns(except) { $$('select').forEach(s => { if (s._dd && s._dd.wrap !== except) s._dd.close(); }); }
document.addEventListener('pointerdown', e => { if (!e.target.closest('.dd')) closeDropdowns(); });
function enhanceSelect(sel) {
  if (sel._dd) return sel._dd.refresh();
  const multi = sel.multiple, id = sel.id + '-dd', wrap = document.createElement('div');
  wrap.className = 'dd' + (multi ? ' multi' : '');
  wrap.innerHTML = `<button type="button" class="dd-btn" id="${id}" aria-haspopup="listbox" aria-expanded="false"><span class="dd-val"></span>${ic('chevron-down', 'dd-chev')}</button>
    <div class="dd-panel" hidden><div class="search-box">${ic('search')}<input class="dd-search" type="search" placeholder="Search…" autocomplete="off"
      role="combobox" aria-expanded="true" aria-controls="${id}-list"></div>
    <div class="dd-count muted" aria-live="polite"></div><div class="dd-list" id="${id}-list" role="listbox"${multi ? ' aria-multiselectable="true"' : ''}></div>
    ${multi ? '<div class="dd-foot"><button type="button" class="btn quiet dd-clear">Clear</button><button type="button" class="btn primary dd-done">Done</button></div>' : ''}</div>`;
  sel.after(wrap);
  sel.hidden = true;
  const btn = wrap.querySelector('.dd-btn'), panel = wrap.querySelector('.dd-panel'), search = wrap.querySelector('.dd-search'), list = wrap.querySelector('.dd-list');
  const label = sel.labels?.[0];
  if (label) label.htmlFor = id;
  const name = (label?.textContent || sel.getAttribute('aria-label') || sel.dataset.label || '').trim();
  btn.setAttribute('aria-label', name);
  search.setAttribute('aria-label', 'Search ' + name.toLowerCase());
  let active = 0;
  const shown = () => { const q = search.value.trim().toLowerCase(); return [...sel.options].filter(o => o.text.toLowerCase().includes(q)); };
  function refresh() {
    const picked = [...sel.selectedOptions];
    const text = multi ? (picked.length ? picked.map(o => o.text).join(', ') : sel.dataset.placeholder || 'Any') : picked[0]?.text || '';
    wrap.querySelector('.dd-val').innerHTML = (sel.dataset.label ? `<small>${esc(sel.dataset.label)}</small>` : '') + esc(text);
    btn.classList.toggle('has-value', multi && picked.length > 0);
    btn.title = text;
  }
  function draw() {
    const s = shown();
    active = Math.max(0, Math.min(active, s.length - 1));
    wrap.querySelector('.dd-count').textContent = `${s.length} of ${sel.options.length}${multi ? ` · ${sel.selectedOptions.length} chosen` : ''}`;
    list.innerHTML = s.length ? s.map((o, i) => `<div role="option" id="${id}-o${o.index}" data-i="${o.index}" aria-selected="${o.selected}" class="dd-opt${i === active ? ' active' : ''}">${multi ? '<span class="dd-box"></span>' : ''}<span>${esc(o.text)}</span></div>`).join('')
      : `<div class="dd-empty">No options match "${esc(search.value.trim())}"</div>`;
    search.setAttribute('aria-activedescendant', s[active] ? `${id}-o${s[active].index}` : '');
  }
  function open() {
    if (!panel.hidden && !panel.classList.contains('leaving')) return;
    closeDropdowns(wrap);
    panel.classList.remove('leaving');
    search.value = '';
    active = Math.max(0, shown().findIndex(o => o.selected));
    draw();
    panel.hidden = false;
    panel.style.left = panel.style.right = '';
    if (panel.getBoundingClientRect().right > innerWidth - 8) { panel.style.left = 'auto'; panel.style.right = '0'; } // keep on screen
    btn.setAttribute('aria-expanded', 'true');
    search.focus();
  }
  function close(focusBtn) {
    if (panel.hidden || panel.classList.contains('leaving')) return;
    btn.setAttribute('aria-expanded', 'false');
    animateOut(panel, 'leaving', () => { panel.hidden = true; });
    if (focusBtn) btn.focus();
  }
  function choose(o) {
    if (multi) o.selected = !o.selected; else sel.value = o.value;
    sel.dispatchEvent(new Event('change', { bubbles: true }));
    refresh();
    if (multi) draw(); else close(true);
  }
  btn.onclick = () => (panel.hidden ? open() : close());
  btn.onkeydown = e => { if (e.key === 'ArrowDown' || e.key === 'ArrowUp') { e.preventDefault(); open(); } };
  search.oninput = () => { active = 0; draw(); };
  search.onkeydown = e => {
    const s = shown();
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
      e.preventDefault();
      active = (active + (e.key === 'ArrowDown' ? 1 : -1) + s.length) % Math.max(1, s.length);
      draw();
      list.querySelector('.active')?.scrollIntoView({ block: 'nearest' });
    } else if (e.key === 'Enter') { e.preventDefault(); if (s[active]) choose(s[active]); }
    else if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); close(true); } // close the list, not the meal picker
    else if (e.key === 'Tab') close();
  };
  list.onclick = e => {
    const el = e.target.closest('[data-i]');
    if (!el) return;
    active = shown().findIndex(o => o.index === +el.dataset.i);
    choose(sel.options[+el.dataset.i]);
  };
  wrap.querySelector('.dd-clear')?.addEventListener('click', () => { [...sel.options].forEach(o => { o.selected = false; }); sel.dispatchEvent(new Event('change', { bubbles: true })); refresh(); draw(); });
  wrap.querySelector('.dd-done')?.addEventListener('click', () => close(true));
  sel._dd = { refresh, close, wrap };
  refresh();
  icons();
}
window.addEventListener('hashchange', () => {

  document.startViewTransition(route).ready.catch(() => {}); // a skipped transition (e.g. rapid clicks) still runs route

async function boot() {
  const lucideReady = new Promise(r => window.lucide ? r() : window.addEventListener('load', r, { once: true }));
  await Promise.race([Promise.all([document.fonts.ready, lucideReady]), new Promise(r => setTimeout(r, 2500))]);
  document.body.classList.remove('booting');
  $('main').removeAttribute('aria-busy');
  render();
  cascadeIn($$(`#${view} :is(${CASCADE})`));
  $(`#${view} h1`)?.focus({ preventScroll: true });
}
